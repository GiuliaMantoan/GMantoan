{{/* Display author list. Co-authors listed in data/coauthors.yaml link to their web pages. */}} {{ $taxonomy := "authors" }} {{ with .Param $taxonomy }} {{ range $index, $value := . }} {{- $profile_page := site.GetPage (printf "/%s/%s" $taxonomy (. | urlize)) -}} {{- $name := $profile_page.Params.name | default ($value|markdownify) -}} {{- $ext := index site.Data.coauthors $value -}} {{- if gt $index 0 }}, {{ end -}} {{- if $ext -}} {{$name}} {{- else if $profile_page -}} {{$name}} {{- else -}} {{$name}} {{- end -}} {{- end -}} {{ end }}
